import cv2
import matplotlib.pyplot as plt
import numpy as np
img = cv2.imread('screenshot.png')
# đổi màu từ BGR sang RGB
cvt = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
# đổi sang ảnh xám
gray = cv2.cvtColor(cvt, cv2.COLOR_RGB2GRAY)
# âm bản
nigga = 255 - img
# độ sáng
bright = cv2.convertScaleAbs(img, alpha=1, beta=50)
# độ tương phản
contrast = cv2.convertScaleAbs(img, alpha=1.5, beta=0)
# thay đổi kích thước ảnh
h,w = img.shape[:2]
resized = cv2.resize(img, (h,w))
# thu phóng ảnh
resized2 = cv2.resize(img, None, fx=0.5, fy=0.5)
# cắt ảnh
cropped = img[0:w, 0:h]
# xoay ảnh
rotated = cv2.getRotationMatrix2D((w/2, h/2), 45, 1)
re = cv2.warpAffine(img, rotated, (w, h))
# input
tx = int(input("Nhập khoảng dịch theo trục x: "))
ty = int(input("Nhập khoảng dịch theo trục y: "))
# dịch ảnh
da = np.float32([[1,0,tx],[0,1,ty]])
# đọc video
cap = cv2.VideoCapture('video.mp4')
while cap.isOpened():
    ret,frame = cap.read()
    cv2.imshow('Video', frame)
    if cv2.waitKey(1) == ord('q'):
        break
cap.release()
cv2.destroyAllWindows()
# kích thước video
width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
# bộ lọc trung bình
blur = cv2.blur(img, (5, 5))
# bộ lọc trung vị
median = cv2.medianBlur(img, 5)
# bộ lọc Bilateral
bilateral = cv2.bilateralFilter(img, 9, 75, 75)
# bộ lọc Gaussian
gauu = cv2.GaussianBlur(img, (5, 5), 0)
# lấy kernel
kernel = np.ones((5,5), np.uint8)
# co giãn ảnh
dilated = cv2.dilate(img, kernel, iterations=1)
eroded = cv2.erode(img, kernel, iterations=1)
# phép đóng mở
closing = cv2.morphologyEx(img, cv2.morphologyEx, kernel, iterations=1)
opening = cv2.morphologyEx(img, cv2.morphologyEx, kernel, iterations=1)
# phân ngưỡng Otsu
ret,th = cv2.threshold(gray, 0,255, cv2.THRESH_BINARY+cv2.THRESH_OTSU)
# phân ngưỡng thích nghi
th2 = cv2.adaptiveThreshold(gray, 255, cv2.ADAPTIVE_THRESH_MEAN_C,cv2.THRESH_BINARY,11,2)
# phân ngưỡn nhị phân
ret,th3 = cv2.threshold(gray,127, 255 ,0)
# tách biên canny
edges = cv2.Canny(gray, 100, 200)
# tách biên Sobel
sobelx = cv2.Sobel(gray, cv2.CV_64F,1,0, 3)
sobely = cv2.Sobel(gray, cv2.CV_64F,0,1, 3)
abs_sobelx = cv2.convertScaleAbs(sobelx)
abs_sobely = cv2.convertScaleAbs(sobely)
Result = cv2.addWeighted(abs_sobelx, 0.5, abs_sobely, 0.5, 0)
# tách biên Laplacian
laplacian = cv2.Laplacian(gray,cv2.CV_64F, 3)
abs_laplacian = cv2.convertScaleAbs(laplacian)
# countours
countours, hierarchy = cv2.findContours(th3, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)
cv2.drawContours(img, countours, -1,(0,255,0), 3)
cv2.imshow('Contours', img)
